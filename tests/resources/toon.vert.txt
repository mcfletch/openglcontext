// 	http://www.lighthouse3d.com/opengl/glsl/index.php?toon2
// By way of:
//  http://www.pygame.org/wiki/GLSLExample
#version 330 core

// Supplied by the render pass to any GLSLObject that names them.
uniform mat4 mat_modelproj;
uniform mat4 itp_modelview;

in vec3 aPosition;
in vec3 aNormal;

out vec3 normal;

void main() {
	normal = mat3( itp_modelview ) * aNormal;
	gl_Position = mat_modelproj * vec4( aPosition, 1.0 );
}
