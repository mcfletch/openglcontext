// Aliased checkboard pattern discarding "white"
// loosely based on orange book chapter 11.3
//
// texCoord and vertexColor come from res://simpleshader_vert_txt, which is
// where a fragment-only GLSLObject gets its vertex stage.
#version 330 core

in vec2 texCoord;
in vec4 vertexColor;
out vec4 fragColor;

void main() {
	if (texCoord.x > .5) {
		if (texCoord.y > .5) {
			fragColor = mix( vertexColor, vec4( 0,0,0,1 ), .5);
		} else {
			discard;
		}
	} else {
		if (texCoord.y > .5) {
			discard;
		} else {
			fragColor = mix( vertexColor, vec4( 0,0,0,1 ), .5);
		}
	}
}
